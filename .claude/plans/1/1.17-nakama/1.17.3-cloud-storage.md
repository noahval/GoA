## Implementation

### 3.1 Generic Storage Functions
Low-level read/write for any data.

**Write Storage**:
```gdscript
func write_storage(collection: String, key: String, value: Dictionary):
	if not is_authenticated:
		DebugLogger.error("Cannot write storage: Not authenticated")
		return null

	var write_object = NakamaWriteStorageObject.new(
		collection,
		key,
		2,  # Read permission: owner only
		1,  # Write permission: owner only
		JSON.stringify(value),
		""
	)

	var result = await client.write_storage_objects_async(session, [write_object])

	if result.is_exception():
		DebugLogger.error("Storage write failed: " + result.get_exception().message)
		return null

	return result
```

**Read Storage**:
```gdscript
func read_storage(collection: String, key: String):
	if not is_authenticated:
		DebugLogger.error("Cannot read storage: Not authenticated")
		return null

	var storage_id = NakamaStorageObjectId.new(collection, key, user_id)
	var result = await client.read_storage_objects_async(session, [storage_id])

	if result.is_exception():
		DebugLogger.error("Storage read failed: " + result.get_exception().message)
		return null

	if result.objects.size() > 0:
		return JSON.parse_string(result.objects[0].value)

	return null
```

### 3.2 Game Save/Load Functions
High-level functions for full game state.

**Save Game**:
```gdscript
func save_game() -> bool:
	if not is_authenticated:
		DebugLogger.error("Cannot save: Not authenticated", "NAKAMA")
		return false  # Don't mark connection lost for auth issues

	var current_time = Time.get_unix_time_from_system()
	var save_data = {
		"version": "2.0",
		"timestamp": current_time,
		"global": _get_global_data(),
		"level1_vars": _get_level1_vars_data()
	}

	var result = await write_storage("player_data", "game_save", save_data)

	if result:
		_mark_connection_success()
		last_successful_save_timestamp = current_time  # For quit warning (see 1.17.6)
		DebugLogger.info("Game saved to cloud", "NAKAMA")
		return true
	else:
		_mark_connection_lost()
		DebugLogger.error("Game save failed", "NAKAMA")
		return false
```

**Load Game**:
```gdscript
func load_game() -> bool:
	if not is_authenticated:
		DebugLogger.error("Cannot load: Not authenticated")
		return false

	var save_data = await read_storage("player_data", "game_save")

	if not save_data:
		DebugLogger.info("No cloud save found")
		return false

	if not save_data.has("global") or not save_data.has("level1_vars"):
		DebugLogger.error("Invalid cloud save structure")
		return false

	_set_global_data(save_data.global)
	_set_level1_vars_data(save_data.level1_vars)

	DebugLogger.info("Game loaded from cloud")
	return true
```

**Note**: Settings are included in the game save using the combined approach from 1.x-settings-panel.md. Settings are preserved during prestige/reset operations through surgical preservation.

### 3.3 Connection State Tracking

```gdscript
func _mark_connection_success():
	var was_disconnected = not is_connected
	is_connected = true

	if was_disconnected:
		connection_restored.emit()
		Global.show_notification("Connection restored", Global.NOTIFICATION_TYPE_SUCCESS)

func _mark_connection_lost():
	if is_connected:
		is_connected = false
		connection_lost.emit()
		Global.show_notification("Connection lost - progress not saved", Global.NOTIFICATION_TYPE_WARNING)
```

## 3.4 Error Handling & Retry Logic

### Network Timeout Handling

```gdscript
const STORAGE_TIMEOUT = 15  # 15 second timeout for storage operations

func write_storage(collection: String, key: String, value: Dictionary):
	if not is_authenticated:
		DebugLogger.error("Cannot write storage: Not authenticated", "NAKAMA")
		return null

	var write_object = NakamaWriteStorageObject.new(
		collection,
		key,
		2,  # Read permission: owner only
		1,  # Write permission: owner only
		JSON.stringify(value),
		""
	)

	# Set timeout for this operation
	var original_timeout = client.timeout
	client.timeout = STORAGE_TIMEOUT

	var result = await client.write_storage_objects_async(session, [write_object])

	# Restore original timeout
	client.timeout = original_timeout

	if result.is_exception():
		var error = result.get_exception().message
		DebugLogger.error("Storage write failed: " + error, "NAKAMA")

		# Classify error for better handling
		if _is_network_error(error):
			_last_error_type = "network"
		elif _is_quota_error(error):
			_last_error_type = "quota"
		else:
			_last_error_type = "unknown"

		return null

	return result
```

### Error Classification

```gdscript
var _last_error_type: String = ""  # "network", "quota", "auth", "unknown"

func _is_network_error(error_message: String) -> bool:
	var network_keywords = ["network", "timeout", "connection", "unreachable", "dns"]
	var lower_error = error_message.to_lower()
	for keyword in network_keywords:
		if keyword in lower_error:
			return true
	return false

func _is_quota_error(error_message: String) -> bool:
	var quota_keywords = ["quota", "limit", "storage full", "size exceeded"]
	var lower_error = error_message.to_lower()
	for keyword in quota_keywords:
		if keyword in lower_error:
			return true
	return false

func _is_auth_error(error_message: String) -> bool:
	var auth_keywords = ["unauthorized", "authentication", "session", "token"]
	var lower_error = error_message.to_lower()
	for keyword in auth_keywords:
		if keyword in lower_error:
			return true
	return false
```

### Save Retry Logic

```gdscript
const MAX_SAVE_RETRIES = 2
const SAVE_RETRY_DELAY = 1.0  # 1 second between retries

func save_game_with_retry() -> bool:
	"""
	Saves game with automatic retry on network failure.
	Returns: true if save succeeded, false otherwise
	"""
	for attempt in range(MAX_SAVE_RETRIES + 1):
		var success = await save_game()

		if success:
			return true

		# Don't retry on auth or quota errors - only network
		if _last_error_type != "network":
			DebugLogger.error("Save failed (non-network error), not retrying", "NAKAMA")
			return false

		# Retry on network errors
		if attempt < MAX_SAVE_RETRIES:
			DebugLogger.info("Save failed, retrying in %d second(s)..." % SAVE_RETRY_DELAY, "NAKAMA")
			await get_tree().create_timer(SAVE_RETRY_DELAY).timeout

	DebugLogger.error("Save failed after %d retries" % MAX_SAVE_RETRIES, "NAKAMA")
	return false
```

### Storage Quota Handling

```gdscript
signal quota_exceeded()  # Emitted when storage quota is full

func save_game() -> bool:
	if not is_authenticated:
		DebugLogger.error("Cannot save: Not authenticated", "NAKAMA")
		return false

	var current_time = Time.get_unix_time_from_system()
	var save_data = {
		"version": "1.0",  # Matches Global.SAVE_VERSION
		"timestamp": current_time,
		"global": _get_global_data(),
		"level1_vars": _get_level1_vars_data()
	}

	var result = await write_storage("player_data", "game_save", save_data)

	if result:
		_mark_connection_success()
		last_successful_save_timestamp = current_time
		DebugLogger.info("Game saved to cloud", "NAKAMA")
		return true
	else:
		# Check error type and handle appropriately
		if _last_error_type == "quota":
			DebugLogger.error("Storage quota exceeded", "NAKAMA")
			quota_exceeded.emit()
			Global.show_notification("Cloud storage full - save failed", Global.NOTIFICATION_TYPE_WARNING)
		elif _last_error_type == "network":
			_mark_connection_lost()
		elif _last_error_type == "auth":
			DebugLogger.error("Authentication expired during save", "NAKAMA")
			Global.show_notification("Session expired - please login again", Global.NOTIFICATION_TYPE_WARNING)
		else:
			_mark_connection_lost()

		DebugLogger.error("Game save failed: " + _last_error_type, "NAKAMA")
		return false
```

### Load Validation

```gdscript
func load_game() -> bool:
	if not is_authenticated:
		DebugLogger.error("Cannot load: Not authenticated", "NAKAMA")
		return false

	var save_data = await read_storage("player_data", "game_save")

	if not save_data:
		DebugLogger.info("No cloud save found", "NAKAMA")
		return false

	# Validate save structure
	if not save_data.has("version"):
		DebugLogger.error("Invalid cloud save: missing version", "NAKAMA")
		return false

	if not save_data.has("global") or not save_data.has("level1_vars"):
		DebugLogger.error("Invalid cloud save: missing data sections", "NAKAMA")
		return false

	# Check version compatibility
	var save_version = save_data.get("version", "1.0")
	if not _is_save_compatible(save_version):
		DebugLogger.error("Incompatible save version: " + save_version, "NAKAMA")
		Global.show_notification("Save file too old - cannot load", Global.NOTIFICATION_TYPE_WARNING)
		return false

	# Load data
	_set_global_data(save_data.global)
	_set_level1_vars_data(save_data.level1_vars)

	DebugLogger.info("Game loaded from cloud (version: " + save_version + ")", "NAKAMA")
	return true

func _is_save_compatible(save_version: String) -> bool:
	# Currently only version 1.0 is supported
	# Future: add migration logic for older versions
	return save_version == "1.0"
```

## 3.5 Automatic Save Timing

```gdscript
# Auto-save configuration
const AUTO_SAVE_INTERVAL = 120.0  # 2 minutes in seconds
var auto_save_timer: Timer = null
var auto_save_enabled: bool = true

func _ready():
	# ... existing code ...

	# Setup auto-save timer
	auto_save_timer = Timer.new()
	auto_save_timer.wait_time = AUTO_SAVE_INTERVAL
	auto_save_timer.timeout.connect(_on_auto_save_timer_timeout)
	add_child(auto_save_timer)

	if auto_save_enabled:
		auto_save_timer.start()

func _on_auto_save_timer_timeout():
	if is_authenticated and not is_loading_cloud_save:
		DebugLogger.info("Auto-save triggered", "NAKAMA")
		await save_game_with_retry()

func set_auto_save_enabled(enabled: bool):
	auto_save_enabled = enabled
	if auto_save_timer:
		if enabled:
			auto_save_timer.start()
		else:
			auto_save_timer.stop()
```

## Notes & Decisions

**Decision 1**: 15 second timeout for storage operations
- **Rationale**: Longer than default (10s) to account for large save files, but not infinite
- **Benefit**: Prevents indefinite hangs on slow connections
- **Trade-off**: Users on very slow connections may see timeouts

**Decision 2**: Retry on network errors only (not auth/quota)
- **Rationale**: Network errors are transient, auth/quota errors require user action
- **Implementation**: Classify errors, only retry if `_last_error_type == "network"`
- **Benefit**: Avoids wasting retries on permanent failures

**Decision 3**: Max 2 retries with 1 second delay
- **Rationale**: Quick retries for transient network glitches, not prolonged outages
- **Alternative considered**: Exponential backoff (rejected - save operations should fail fast)
- **Total wait**: 2 seconds max for save operation (acceptable)

**Decision 4**: Auto-save every 2 minutes
- **Rationale**: Balances data safety with server load
- **Configurable**: Can be disabled if user prefers manual saves
- **Alternative considered**: 5 minutes (rejected - too infrequent), 30 seconds (rejected - too frequent, server load)

**Decision 5**: Save version "1.0" (matches Global.SAVE_VERSION)
- **Rationale**: Save format version tracks save structure, not game version (0.1)
- **Future**: Increment when save structure changes, add migration logic
- **Benefit**: Clear separation of concerns

**Decision 6**: Validate save structure before loading
- **Rationale**: Prevents crashes from corrupted/malformed saves
- **Checks**: Version field, global section, level1_vars section
- **Benefit**: Graceful degradation instead of errors

**Decision 7**: Emit specific signals for different error types
- **Rationale**: UI can respond differently to network vs quota vs auth errors
- **Signals**: `connection_lost` (network), `quota_exceeded` (storage full)
- **Benefit**: Better user feedback

**Decision 8**: Store timeout setting on client object
- **Rationale**: Nakama client has built-in timeout property
- **Implementation**: Set before operation, restore after
- **Benefit**: Simple, no need for custom timeout logic
